Модуль под Telegram-систему «несколько ботов через общую БД» · Python · aiogram 3.x · SQLAlchemy 2.0 · PostgreSQL

Задача. В многоботовой системе (Garant / Portal / Admin) один бот выдаёт ссылку-приглашение в закрытый чат сделки, другой её гасит. Ссылка должна срабатывать ровно один раз, быть привязана к конкретному user_id и иметь срок жизни. Это типовое «узкое место» таких систем — ниже рабочая, проверенная реализация именно этого узла.

Ядро: атомарное погашение (race-safe)

# выигрывает ровно один параллельный /start — двойной вход невозможен
stmt = (
    update(AccessLink)
    .where(
        AccessLink.token_hash == token_hash,
        AccessLink.redeemed_at.is_(None),
    )
    .values(redeemed_at=now, redeemed_by=user_id)
)
res = s.execute(stmt)
s.commit()
return RedeemResult(True, "ok") if res.rowcount == 1 else RedeemResult(False, "already_used")

Подключение в aiogram 3.x (Portal Bot)

@router.message(CommandStart(deep_link=True))
async def on_start(message, command):
    res = svc.redeem(command.args, user_id=message.from_user.id)
    if res.ok:
        # res.resource == "chat:-100…" → выдаём доступ
        await message.answer("Доступ открыт ✅")
    else:
        await message.answer(_human(res.reason))  # понятная причина отказа
